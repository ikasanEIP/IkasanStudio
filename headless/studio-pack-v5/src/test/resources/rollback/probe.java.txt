    public final java.util.concurrent.CountDownLatch rolledBack = new java.util.concurrent.CountDownLatch(1);
    public final java.util.concurrent.CountDownLatch allowRetry = new java.util.concurrent.CountDownLatch(1);
    public final java.util.concurrent.atomic.AtomicInteger attempts = new java.util.concurrent.atomic.AtomicInteger();
    public volatile int completionStatus = -1;
    public volatile boolean redelivered;
    public volatile String firstMessageId;
    public volatile String retryMessageId;

    private void injectCommitFailure(Object value) {
        int attempt = attempts.incrementAndGet();
        if (attempt == 1) {
            if (!org.springframework.transaction.support.TransactionSynchronizationManager.isActualTransactionActive())
                throw new IllegalStateException("Rollback fixture requires an actual transaction");
            if (value instanceof jakarta.jms.Message message) {
                try { firstMessageId = message.getJMSMessageID(); }
                catch (jakarta.jms.JMSException failure) { throw new RuntimeException(failure); }
            }
            org.springframework.transaction.support.TransactionSynchronizationManager.registerSynchronization(
                new org.springframework.transaction.support.TransactionSynchronization() {
                    @Override public void beforeCommit(boolean readOnly) {
                        throw new CommitFailure();
                    }
                    @Override public void afterCompletion(int status) {
                        completionStatus = status;
                        rolledBack.countDown();
                    }
                });
        } else if (attempt == 2 && value instanceof jakarta.jms.Message message) {
            try {
                redelivered = message.getJMSRedelivered();
                retryMessageId = message.getJMSMessageID();
                if (!allowRetry.await(20, java.util.concurrent.TimeUnit.SECONDS))
                    throw new IllegalStateException("Test did not release retry within 20 seconds");
            } catch (Exception failure) { throw new RuntimeException(failure); }
        }
    }
